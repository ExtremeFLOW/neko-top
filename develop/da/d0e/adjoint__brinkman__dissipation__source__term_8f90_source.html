153 !! @param f_x, f_y, f_z the RHS of the adjoint

155 !! @param K a scale
156 !! @param u, v, w the velocity fields of the primal

173 type(field_t), pointer, intent(in) :: f_x, f_y, f_z
174 class(design_t), intent(in), target :: design

251 !> Computes the source term and adds the result to `fields`.
252 !! @param this The source term.
253 !! @param time The time state.
254 subroutine adjoint_brinkman_dissipation_source_term_compute(this, time)
255 class(adjoint_brinkman_dissipation_source_term_t), intent(inout) :: this
256 type(time_state_t), intent(in) :: time
257 type(field_t), pointer :: fu, fv, fw
258 type(field_t), pointer :: work
259 integer :: temp_indices(1)
260 type(field_t), pointer :: accumulate, fld_gl, chi_gl
261 integer :: temp_indices_gl(3)
262 integer :: n_gl, nel
263
264 ! Off outside the objective's time window
265 if (abscmp(this%scale, 0.0_rp)) return
266
267 fu => this%fields%get_by_index(1)
268 fv => this%fields%get_by_index(2)
269 fw => this%fields%get_by_index(3)
270
271 ! BE SO CAREFUL HERE
272 ! It make look the same as the Brinkman term, but it's assumed that
273 ! this source term acts on the adjoint, and the u,v,w here come from
274 ! the primal
275 call neko_scratch_registry%request_field(work, temp_indices(1), .false.)
276 call field_copy(work, this%chi)
277
278 ! scale by K and volume
279 call field_cmult(work, this%scale * this%K / this%volume)
280
281 ! mask
282 if (this%if_mask) then
283 call mask_exterior_const(work, this%mask, 0.0_rp)
284 end if
285
286 if (this%dealias) then
287 nel = this%coef%msh%nelv
288 n_gl = nel * this%Xh_GL%lxyz
289 call this%scratch_GL%request_field(accumulate, temp_indices_gl(1), &
290 .false.)
291 call this%scratch_GL%request_field(fld_gl, temp_indices_gl(2), .false.)
292 call this%scratch_GL%request_field(chi_gl, temp_indices_gl(3), .false.)
293
294 call this%GLL_to_GL%map(chi_gl%x, work%x, nel, this%Xh_GL)
295
296 ! u
297 call this%GLL_to_GL%map(fld_gl%x, this%u%x, nel, this%Xh_GL)
298 call field_col3(accumulate, chi_gl, fld_gl)
299 if (neko_bcknd_device .eq. 1) then
300 call device_col2(accumulate%x_d, this%c_Xh_GL%B_d, n_gl)
301 call this%GLL_to_GL%map(work%x, accumulate%x, nel, this%Xh_GLL)
302 call device_invcol2(work%x_d, this%coef%B_d, work%size())
303 else
304 call col2(accumulate%x, this%c_Xh_GL%B, n_gl)
305 call this%GLL_to_GL%map(work%x, accumulate%x, nel, this%Xh_GLL)
306 call invcol2(work%x, this%coef%B, work%size())
307 end if
308 call field_add2(fu, work)
309
310 ! v
311 call this%GLL_to_GL%map(fld_gl%x, this%v%x, nel, this%Xh_GL)
312 call field_col3(accumulate, chi_gl, fld_gl)
313 if (neko_bcknd_device .eq. 1) then
314 call device_col2(accumulate%x_d, this%c_Xh_GL%B_d, n_gl)
315 call this%GLL_to_GL%map(work%x, accumulate%x, nel, this%Xh_GLL)
316 call device_invcol2(work%x_d, this%coef%B_d, work%size())
317 else
318 call col2(accumulate%x, this%c_Xh_GL%B, n_gl)
319 call this%GLL_to_GL%map(work%x, accumulate%x, nel, this%Xh_GLL)
320 call invcol2(work%x, this%coef%B, work%size())
321 end if
322 call field_add2(fv, work)
323
324 ! w
325 if (this%gdim .eq. 3) then
326 call this%GLL_to_GL%map(fld_gl%x, this%w%x, nel, this%Xh_GL)
327 call field_col3(accumulate, chi_gl, fld_gl)
328 if (neko_bcknd_device .eq. 1) then
329 call device_col2(accumulate%x_d, this%c_Xh_GL%B_d, n_gl)
330 call this%GLL_to_GL%map(work%x, accumulate%x, nel, this%Xh_GLL)
331 call device_invcol2(work%x_d, this%coef%B_d, work%size())
332 else
333 call col2(accumulate%x, this%c_Xh_GL%B, n_gl)
334 call this%GLL_to_GL%map(work%x, accumulate%x, nel, this%Xh_GLL)
335 call invcol2(work%x, this%coef%B, work%size())
336 end if
337 call field_add2(fw, work)
338 end if
339
340 call this%scratch_GL%relinquish_field(temp_indices_gl)
341
342 else
343 ! multiple and add the RHS
344 call field_addcol3(fu, this%u, work)
345 call field_addcol3(fv, this%v, work)
346 if (this%gdim .eq. 3) then
347 call field_addcol3(fw, this%w, work)
348 end if
349
350 end if
351
352 call neko_scratch_registry%relinquish_field(temp_indices)
353
354 end subroutine adjoint_brinkman_dissipation_source_term_compute
