118 !> The common constructor using a JSON object.
119 !! @param this the constraint.
120 !! @param json the JSON object.
121 !! @param design the design.
122 !! @param simulation the simulation.
123 subroutine volume_constraint_init_json_sim(this, json, design, simulation)
124 class(volume_constraint_t), intent(inout) :: this
125 type(json_file), intent(inout) :: json
126 class(design_t), intent(in) :: design
127 type(simulation_t), target, intent(inout) :: simulation
128
129 character(len=:), allocatable :: mask_name
130 character(len=:), allocatable :: name
131 logical :: is_max
132 real(kind=rp) :: limit
133
134 call json_get_or_default(json, "name", name, "Volume constraint")
135 call json_get_or_default(json, "mask_name", mask_name, "")
136 call json_get_or_default(json, "is_max", is_max, .false.)
137 call json_get(json, "limit", limit)
138
139 call this%init_from_components(design, simulation, name, mask_name, &
140 is_max, limit)
141
142 ! check if we have a mapping
143 if ('mapping' .in. json) then
144 ! Initialize the mapper
145 call this%mapping%init_base(this%c_Xh)
146 call this%mapping%add(json, 'mapping')
147 this%if_mapping = .true.
148 ! recompute value after mapping
149 call this%update_value(design)
150 end if
151
152 end subroutine volume_constraint_init_json_sim

154 !> The direct initializer from attributes.
155 !! @param this the constraint.
156 !! @param design the design.
157 !! @param simulation the simulation.
158 !! @param name the name of the constraint.
159 !! @param mask_name the name of the mask.
160 !! @param is_max whether it is a maximum volume constraint.
161 !! @param limit The volume limit value.
162 subroutine volume_constraint_init_from_components(this, design, simulation, &
163 name, mask_name, is_max, limit)
164 class(volume_constraint_t), intent(inout) :: this
165 class(design_t), intent(in) :: design
166 type(simulation_t), target, intent(inout) :: simulation
167 character(len=*), intent(in) :: mask_name
168 character(len=*), intent(in) :: name
169 logical, intent(in) :: is_max
170 real(kind=rp), intent(in) :: limit
171
172 type(field_t), pointer :: work
173 integer :: temp_indices(1)
174
175 call this%free()
176
177 ! Initialize the base class
178 call this%init_base(name, design%size(), mask_name)
179
180 ! Store the attributes
181 this%is_max = is_max
182 this%limit = limit
183 this%c_Xh => simulation%neko_case%fluid%c_Xh
184
185 ! Now we can extract the mask/has_mask from the design
186 if (this%has_mask) then
187
188 ! calculate the volume of the optimization domain
189 call neko_scratch_registry%request(work, temp_indices(1), .false.)
190 call field_rone(work)
191 call mask_exterior_const(work, this%mask, 0.0_rp)
192
193 if (neko_bcknd_device .eq. 1) then
194 this%volume_domain = device_glsc2(work%x_d, this%c_xh%B_d, &
195 work%size())
196 else
197 this%volume_domain = glsc2_mask(work%x, this%c_Xh%B, &
198 design%size(), this%mask%mask%get(), this%mask%size)
199 end if
200
201 call neko_scratch_registry%relinquish(temp_indices)
202 else
203 this%volume_domain = this%c_Xh%volume
204 end if
205
206 ! ------------------------------------------------------------------------ !
207 ! Initialize the value of constraint
208
209 call this%update_value(design)
210
211 ! ------------------------------------------------------------------------ !
212 ! Initialize the sensitivity value
213 call vector_cfill(this%sensitivity, -1.0_rp / this%volume_domain)
214 if (this%is_max) then
215 call vector_cmult(this%sensitivity, -1.0_rp)
216 end if
217
218 if (this%has_mask) then
219 call mask_exterior_const(this%sensitivity, this%mask, 0.0_rp)
220 end if
221
222 end subroutine volume_constraint_init_from_components
223

230 this%if_mapping = .false.
231 nullify(this%c_Xh)

236 !> The computation of the constraint.
237 !! @param this the constraint
238 !! @param design the design
239 subroutine volume_constraint_update_value(this, design)
240 class(volume_constraint_t), intent(inout) :: this
241 class(design_t), intent(in) :: design
242 real(kind=rp) :: volume
243
244 volume = this%compute_volume(design)
245
246 ! Compute the distance to the target volume
247 this%value = this%limit - volume / this%volume_domain
248
249 ! Invert the sign if it is a maximum constraint
250 if (this%is_max) this%value = -this%value
251
252 end subroutine volume_constraint_update_value
253
254 !> The computation of the sensitivity.
255 !! @param this the constraint
256 !! @param design the design
257 subroutine volume_constraint_update_sensitivity(this, design)
258 class(volume_constraint_t), intent(inout) :: this
259 class(design_t), intent(in) :: design
260 !> Sensitivity field
261 type(field_t), pointer :: unmapped, mapped
262 integer :: idx(2)
263
264 if (this%if_mapping) then
265 ! Recompute and map backward
266 call neko_scratch_registry%request(unmapped, idx(1), .false.)
267 call neko_scratch_registry%request(mapped, idx(2), .false.)
268 call field_cfill(unmapped, -1.0_rp / this%volume_domain)
269 if (this%is_max) then
270 call field_cmult(unmapped, -1.0_rp)
271 end if
272 ! mask
273 if (this%has_mask) then
274 call mask_exterior_const(unmapped, this%mask, 0.0_rp)
275 end if
276 ! map backwards
277 call this%mapping%apply_backward(mapped, unmapped)
278 if (this%has_mask) then
279 call mask_exterior_const(mapped, this%mask, 0.0_rp)
280 end if
281 call field_to_vector(this%sensitivity, mapped)
282
283 call neko_scratch_registry%relinquish(idx)
284
285 else
286 ! Sensitivity is just a constant so it should not be updated
287 end if
288
289 end subroutine volume_constraint_update_sensitivity

294 !> Computes the volume of the design.
295 !!
296 !! Automatically select which design type, or throw an error.
297 !! @param this the constraint.
298 !! @param design the design.
299 function compute_volume(this, design) result(volume)
300 class(volume_constraint_t), intent(inout) :: this
301 class(design_t), intent(in) :: design
302 real(kind=rp) :: volume
303
304 volume = 0.0_rp
305 select type (design)
306 type is (brinkman_design_t)
307 volume = volume_brinkman_design(this, design)
308
309 class default
310 call neko_error('Volume constraint only works with brinkman_design')
311 end select
312
313 end function compute_volume

315 !> Computes the volume of the brinkman_design.
316 !! @param this the constraint.
317 !! @param design the design.
318 function volume_brinkman_design(this, design) result(volume)
319 class(volume_constraint_t), intent(inout) :: this
320 type(brinkman_design_t), intent(in) :: design
321 real(kind=rp) :: volume
322 type(field_t), pointer :: work
323 type(vector_t), pointer :: values, unmapped_values
324 integer :: temp_indices, ind_value, ind_um_value
325
326 call neko_scratch_registry%request(values, ind_value, design%size(), &
327 .false.)
328
329 volume = 0.0_rp
330 if (this%if_mapping) then
331 call neko_scratch_registry%request(unmapped_values, ind_um_value, &
332 design%size(), .false.)
333 call design%get_values(unmapped_values)
334 call this%mapping%apply_forward(values, unmapped_values)
335 call neko_scratch_registry%relinquish(ind_um_value)
336 else
337 call design%get_values(values)
338 end if
339
340 if (this%has_mask) then
341
342 if (neko_bcknd_device .eq. 1) then
343 call neko_scratch_registry%request(work, temp_indices, .false.)
344 call device_copy(work%x_d, values%x_d, design%size())
345 call mask_exterior_const(work, this%mask, 0.0_rp)
346
347 volume = device_glsc2(work%x_d, this%c_xh%B_d, design%size())
348
349 call neko_scratch_registry%relinquish(temp_indices)
350 else
351 volume = glsc2_mask(values%x, this%c_Xh%B, design%size(), &
352 this%mask%mask%get(), this%mask%size)
353 end if
354
355 else
356
357 if (neko_bcknd_device .eq. 1) then
358 volume = device_glsc2(values%x_d, this%c_xh%B_d, design%size())
359 else
360 volume = glsc2(values%x, this%c_Xh%B, design%size())
361 end if
362
363 end if
364
365 call neko_scratch_registry%relinquish(ind_value)
366
367 end function volume_brinkman_design

395 !> Populate log values for volume constraint.
396 !! @param[in] this The constraint object.
397 !! @param[out] values Values corresponding to the log headers.
398 subroutine volume_constraint_get_log_values(this, values)
399 class(volume_constraint_t), intent(in) :: this
400 real(kind=rp), intent(out) :: values(:)
401 real(kind=rp) :: volume_ratio
402
403 if (size(values) .lt. 1) return
404 if (this%is_max) then
405 volume_ratio = this%limit + this%value
406 else
407 volume_ratio = this%limit - this%value
