288 !> Base initializer for the optimizer
289 !! @param this The optimizer object.
290 !! @param optimizer_type The type of the optimizer.
291 !! @param max_iterations The maximum number of iterations.
292 !! @param max_runtime The maximum runtime in seconds.
293 !! @param stop_design_change The stopping criteria for design change.
294 !! @param restart_file The checkpoint file to restart from.
295 !! @param checkpoint_path The path for saving checkpoint files.
296 !! @param checkpoint_base The base name for checkpoint files.
297 !! @param checkpoint_format The file format for checkpoint files.
298 !! @param checkpoint_interval The interval for saving checkpoints in
299 !! iterations.
300 !! @param checkpoint_overwrite Whether to overwrite existing checkpoint files.
301 subroutine optimizer_init_base(this, optimizer_type, max_iterations, &
302 max_runtime, stop_design_change, restart_file, checkpoint_path, &
303 checkpoint_base, checkpoint_format, checkpoint_interval, &
304 checkpoint_overwrite)
305 class(optimizer_t), intent(inout) :: this
306 character(len=*), intent(in) :: optimizer_type
307 integer, intent(in) :: max_iterations
308 real(kind=rp), intent(in), optional :: max_runtime
309 real(kind=rp), intent(in), optional :: stop_design_change
310 character(len=*), intent(in), optional :: restart_file
311 character(len=*), intent(in), optional :: checkpoint_path
312 character(len=*), intent(in), optional :: checkpoint_base
313 character(len=*), intent(in), optional :: checkpoint_format
314 integer, intent(in), optional :: checkpoint_interval
315 logical, intent(in), optional :: checkpoint_overwrite
316 character(len=LOG_SIZE) :: msg
317 character(len=256) :: file_name
318 integer :: i
319 logical :: exists
320
321 ! Initialize internals
322 this%start_time = mpi_wtime()
323
324 ! Mandatory settings
325 this%optimizer_type = optimizer_type
326 this%max_iterations = max_iterations
327
328 ! Optional settings
329 if (present(max_runtime)) this%max_runtime = max_runtime
330 if (present(stop_design_change)) this%stop_design_change = stop_design_change
331 if (present(restart_file)) this%restart_file = restart_file
332 if (present(checkpoint_path)) this%checkpoint_path = checkpoint_path
333 if (present(checkpoint_base)) this%checkpoint_base = checkpoint_base
334 if (present(checkpoint_format)) this%checkpoint_format = checkpoint_format
335 if (present(checkpoint_interval)) then
336 this%checkpoint_interval = checkpoint_interval
337 end if
338 if (present(checkpoint_overwrite)) then
339 this%checkpoint_overwrite = checkpoint_overwrite
340 end if
341
342 if (this%max_runtime .gt. 0.0_rp) then
343 write(msg, '(A,F0.0,A)') 'Optimizer will stop after: ', &
344 this%max_runtime, ' seconds.'
345 call neko_log%message(msg)
346
347 ! Look up the runtime checkpoint file if no restart file is specified
348 if (len_trim(this%restart_file) .eq. 0) then
349 exists = .false.
350 file_name = checkpoint_filename(this, &
351 basename = "optimizer_rt_checkpoint")
352 if (pe_rank .eq. 0) inquire(file = file_name, exist = exists)
353 call mpi_bcast(exists, 1, mpi_logical, 0, neko_comm)
354 if (exists) this%restart_file = file_name
355 end if
356 end if
357
358 ! If no restart file have been specified, look for the latest checkpoint
359 ! file in the path. This helps us recover from a crash and automatic restart
360 if (len_trim(this%restart_file) .eq. 0) then
361 i = -1
362 if (pe_rank .eq. 0) then
363 do i = this%max_iterations, 1, -1
364 exists = .false.
365 file_name = checkpoint_filename(this, iteration = i)
366 inquire(file = file_name, exist = exists)
367 if (exists) exit
368 end do
369 if (.not. exists) i = -1
370 end if
371
372 call mpi_bcast(i, 1, mpi_integer, 0, neko_comm)
373 if (i .gt. 0) then
374 this%restart_file = checkpoint_filename(this, iteration = i)
375 end if
376 end if
377
378 this%initialized = .true.
379 end subroutine optimizer_init_base

455 ! -------------------------------------------------------------------------- !
456 ! Optimization loop routine
457
458 !> Define the optimization loop
459 !! This subroutine runs the optimization loop until convergence
460 !! or the maximum number of iterations is reached.
461 !!
462 !! The optimization loop can be terminated based on a maximum runtime. In this
463 !! case, a cumulative average is used to determine if the next iteration would
464 !! exceed the maximum runtime.
465 !!
466 !! When restarting from a checkpoint, the continuation parameters are set to
467 !! their values at the checkpointed iteration before the problem is
468 !! re-evaluated.
469 !!
470 !! @param this The optimizer object.
471 !! @param problem The problem object.
472 !! @param design The design object.
473 !! @param simulation The simulation object.
474 subroutine optimizer_run(this, problem, design, simulation)
475 class(optimizer_t), intent(inout) :: this
476 class(problem_t), intent(inout) :: problem
477 class(design_t), intent(inout) :: design
478 type(simulation_t), optional, intent(inout) :: simulation
479 real(kind=rp) :: iteration_time
480 logical :: converged
481 integer :: stop_flag
482
483 ! Initialize variables
484 stop_flag = 1
485 converged = .false.
486
487 if (len_trim(this%restart_file) .gt. 0) then
488 call this%load_checkpoint(design, this%restart_file)
489 end if
490
491 ! compute potential internals for for a potential restart
492 if (this%current_iteration .ne. 0) then
493 call design%set_output_counter(this%current_iteration - 1)
494 if (present(simulation)) then
495 call simulation%set_output_counter(this%current_iteration - 1)
496 end if
497 end if
498
499 ! On a restart, restore the continuation parameters of the checkpointed
500 ! iteration, so that the problem is re-evaluated with the same values as
501 ! in the original run. A fresh start has nothing to restore, and update(0)
502 ! would index the parameter values at 0.
503 if (this%current_iteration .gt. 0) then
504 call nekotop_continuation%update(this%current_iteration)
505 end if
506
507 ! Prepare the problem state before starting the optimization
508 if (present(simulation)) then
509 call simulation%set_design_iteration(this%current_iteration)
510 end if
511 call this%initialize(problem, design, simulation)
512
513 ! Log the initial state of the problem.
514 if (this%current_iteration .eq. 0) then
515 call this%write(this%current_iteration, problem)
516 end if
517
518 ! Save the current design state.
519 call design%write(this%current_iteration)
520
521 call neko_log%section('Optimization Loop')
522
523 do while (this%current_iteration .lt. this%max_iterations)
524 this%current_iteration = this%current_iteration + 1
525 if (pe_rank .eq. 0) then
526 write(*,*) 'Starting iteration ', this%current_iteration
527 end if
528
529 call profiler_start_region('Optimizer iteration')
530 iteration_time = mpi_wtime()
531
532 ! Update the parameters in continuation scheduler
533 call nekotop_continuation%update(this%current_iteration)
534
535 if (present(simulation)) then
536 call simulation%set_design_iteration(this%current_iteration)
537 end if
538 converged = this%step(this%current_iteration, problem, design, &
539 simulation)
540
541 iteration_time = mpi_wtime() - iteration_time
542 call profiler_end_region('Optimizer iteration')
543
544 ! Log the progress and outputs
545 call this%write(this%current_iteration, problem)
546 call design%write(this%current_iteration)
547
548 ! Save checkpoint if enabled
549 if (this%checkpoint_interval .gt. 0 .and. &
550 mod(this%current_iteration, this%checkpoint_interval) == 0) then
551 call this%save_checkpoint(design)
552 end if
553
554 ! --------------------------------------------------------------------- !
555 ! Check stopping criteria
556
557 if (converged) then
558 stop_flag = 0
559 exit
560 else if (this%current_iteration .ge. this%max_iterations) then
561 stop_flag = 1
562 exit
563 else if (this%max_design_change .lt. this%stop_design_change) then
564 stop_flag = 2
565 exit
566 else if (this%out_of_time(iteration_time)) then
567 block
568 character(len=256) :: filename
569 filename = checkpoint_filename(this, &
570 basename = "optimizer_rt_checkpoint")
571 call this%save_checkpoint(design, .true., filename)
572 end block
573 stop_flag = 3
574 exit
575 end if
576 end do
577
578 ! Check that the final design is valid
579 call this%validate(problem, design)
580 call this%print_status(stop_flag, this%current_iteration)
581
582 call neko_log%end_section()
583
584 end subroutine optimizer_run

599 !> Print status message
600 !! Supported flags:
601 !! 0: Converged successfully (SUCCESS)
602 !! 1: Did not converge in max iterations (WARNING)
603 !! 2: Stopped due to design change below threshold (SUCCESS)
604 !! 3: Stopped after reaching max runtime (ERROR)
605 !! *: Unknown reason (FATAL_ERROR)
606 !!
607 !! @param this The optimizer object.
608 !! @param stop_flag The stopping flag.
609 !! @param iteration The number of iterations performed.
610 subroutine optimizer_print_status(this, stop_flag, iteration)
611 class(optimizer_t), intent(in) :: this
612 integer, intent(in) :: stop_flag
613 integer, intent(in) :: iteration
614 character(len=LOG_SIZE) :: msg
615
616 select case (stop_flag)
617 case (0)
618 write(msg, '(A,I0,A)') 'Optimizer converged successfully after ', &
619 iteration, ' iterations.'
620 call neko_log%message(msg)
621 case (1)
622 write(msg, '(A,I0,A)') 'Optimizer did not converge in ', &
623 this%max_iterations, ' iterations.'
624 call neko_log%warning(msg)
625 case (2)
626 write(msg, '(A,A,F8.2,A)') 'Optimizer stopped, design change ', &
627 'below threshold of: ', this%stop_design_change, '.'
628 call neko_log%warning(msg)
629 case (3)
630 if (pe_rank .eq. 0) then
631 write(error_unit, '(A)') &
632 '*** Error: Optimizer stopped due to runtime limit. ***'
633 end if
634
635 case default
636 write(msg, '(A)') 'Optimizer stopped for an unknown reason.'
637 call neko_error(msg)
638 end select
639 end subroutine optimizer_print_status

641 !> Estimate if we are out of time.
642 !! This function uses a cumulative average of iteration times to
643 !! estimate if the next iteration would exceed the maximum runtime.
644 !! @param this The optimizer object.
645 !! @param step_time The time taken for the latest iteration.
646 !! @return out_of_time Logical indicating if we are out of time.
647 function optimizer_out_of_time(this, step_time) result(out_of_time)
648 class(optimizer_t), intent(inout) :: this
649 real(kind=rp), intent(in) :: step_time
650 logical :: out_of_time
651 real(kind=rp) :: elapsed, step, old_avg_weight
652
653 out_of_time = .false.
654 if (this%max_runtime .lt. 0.0_rp) then
655 return
656 end if
657
658 call mpi_allreduce(step_time, step, 1, mpi_real_precision, mpi_max, &
659 neko_comm)
660
661 this%step_count = this%step_count + 1.0_rp
662 old_avg_weight = (this%step_count - 1.0_rp) / this%step_count
663
664 ! Estimate Cumulative Average iteration step
665 this%average_time = step / this%step_count + &
666 this%average_time * old_avg_weight
667
668 ! Determine if next iteration would exceed max runtime
669 elapsed = mpi_wtime() - this%start_time
670 out_of_time = (elapsed + this%average_time) .gt. 0.99_rp * this%max_runtime
671
672 end function optimizer_out_of_time

751 subroutine optimizer_write_log(this, iter, problem, extra_values)

794 ! design change entries

798 nekotop_continuation%params(i)%target

808 !> Construct a checkpoint filename from path/basename/format, applying the
809 !! optimizer's stored `checkpoint_path`/`checkpoint_base`/`checkpoint_format`
810 !! defaults for any component not given. Shared by `optimizer_load_checkpoint`
811 !! and `optimizer_run`'s restart logic so the default path is built in one
812 !! place.
813 !! @param this The optimizer object.
814 !! @param path The path where the checkpoint file is located.
815 !! @param basename The base name of the checkpoint file.
816 !! @param format The file format of the checkpoint file.
817 !! @param iteration The iteration number.
818 function checkpoint_filename(this, path, basename, format, iteration) &
819 result(file_full)
820 class(optimizer_t), intent(in) :: this
821 character(len=*), intent(in), optional :: path
822 character(len=*), intent(in), optional :: basename
823 character(len=*), intent(in), optional :: format
824 integer, intent(in), optional :: iteration
825 character(len=256) :: file_full
826 character(len=:), allocatable :: file_format
827 character(len=256) :: file_path, file_base, file_ext
828
829 ! Default, read from the optimizer's stored checkpoint settings
830 file_path = trim(this%checkpoint_path)
831 file_base = trim(this%checkpoint_base)
832 file_format = trim(this%checkpoint_format)
833
834 ! Override with any provided arguments
835 if (present(path)) file_path = trim(path)
836 if (present(basename)) file_base = trim(basename)
837 if (present(format)) file_format = trim(format)
838
839 ! Ensure the path ends with a '/'
840 if (len_trim(file_path) .eq. 0) then
841 file_path = './'
842 else if (file_path(len_trim(file_path):len_trim(file_path)) .ne. '/') then
843 file_path = trim(file_path) // '/'
844 end if
845
846 select case (trim(file_format))
847 case ('h5', 'hdf5', 'hf5', 'hdf')
848 file_ext = 'h5'
849 case default
850 call neko_error('optimizer: Unsupported checkpoint format: "' // &
851 trim(file_format) // '"')
852 end select
853
854 ! Construct the full filename based on overwrite flag
855 if (present(iteration)) then
856 write(file_full, '(3A,I5.5,2A)') trim(file_path), &
857 trim(file_base), "_", iteration, ".", trim(file_ext)
858 else
859 write(file_full, '(4A)') &
860 trim(file_path), trim(file_base), ".", trim(file_ext)
861 end if
862
863 end function checkpoint_filename

865 !> Save the optimizer checkpoint to a file.
866 !! @param this The optimizer object.
867 !! @param design The design object.
868 !! @param overwrite Whether to overwrite the file if it exists.
869 !! @param filename The name of the file to save the checkpoint to.
870 !! @param path The path where the checkpoint file will be saved.
871 !! @param basename The base name of the file to save the checkpoint.
872 !! @param format The file format to use for the checkpoint file.
873 !! @param iteration The iteration used to construct the checkpoint filename.
874 subroutine optimizer_save_checkpoint(this, design, overwrite, filename, &
875 path, basename, format, iteration)
876 class(optimizer_t), intent(inout) :: this
877 class(design_t), intent(inout) :: design
878 logical, intent(in), optional :: overwrite
879 character(len=*), intent(in), optional :: filename
880 character(len=*), intent(in), optional :: path
881 character(len=*), intent(in), optional :: basename
882 character(len=*), intent(in), optional :: format
883 integer, intent(in), optional :: iteration
884 character(len=256) :: file_full, file_path
885 character(len=12) :: suffix
886 logical :: exists, overwrite_flag
887
888 call neko_log%section('Optimizer save checkpoint')
889 call profiler_start_region('Optimizer save checkpoint')
890
891 ! Ensure support for external iterators.
892 if (present(iteration)) this%current_iteration = iteration
893 if (.not. present(overwrite)) then
894 overwrite_flag = this%checkpoint_overwrite
895 else
896 overwrite_flag = overwrite
897 end if
898
899 ! Create the fill filename and extract relevant components
900 if (present(filename) .and. (present(path) .or. present(basename) .or. &
901 present(format) .or. present(iteration))) then
902 call neko_error('optimizer: Cannot specify both filename and ' // &
903 'path/basename/format for checkpoint loading.')
904 else if (present(filename)) then
905 file_full = trim(filename)
906 else
907 file_full = checkpoint_filename(this, path, basename, format, &
908 this%current_iteration)
909 end if
910
911 ! Throw an error if the file already exists
912 if (pe_rank .eq. 0) then
913 inquire(file = file_full, exist = exists)
914 if (exists .and. .not. overwrite_flag) then
915 call neko_error('optimizer: Checkpoint file "' // &
916 trim(file_full) // '" already exists and overwrite is false.')
917 end if
918 end if
919
920 ! Extract the path and suffix from the full filename.
921 call filename_path(trim(file_full), file_path)
922 call filename_suffix(trim(file_full), suffix)
923
924 ! Create the path if it does not exist
925 exists = .false.
926 if (pe_rank .eq. 0) inquire(file=file_path, exist=exists)
927 call mpi_bcast(exists, 1, mpi_logical, 0, neko_comm)
928 if (.not. exists) then
929 if (pe_rank .eq. 0) call mkdir(trim(file_path))
930 call mpi_barrier(neko_comm)
931 end if
932
933 call neko_log%message('Save general optimizer components')
934 select case (trim(suffix))
935 case ('h5', 'hdf5', 'hf5')
936 call optimizer_save_checkpoint_hdf5(this, file_full, overwrite_flag)
937 case default
938 call neko_error('optimizer: Unsupported checkpoint format: "' // &
939 trim(suffix) // '"')
940 end select
941
942 call neko_log%message('Saving components of ' // this%optimizer_type)
943 call this%save_checkpoint_components(file_full, overwrite_flag)
944
945 call neko_log%message('Save design checkpoint')
946 call design%save_checkpoint(file_full, overwrite_flag)
947
948 call profiler_end_region('Optimizer save checkpoint')
949 call neko_log%end_section()
950
951 end subroutine optimizer_save_checkpoint

953 !> Load the optimizer checkpoint from a file based on file suffix.
954 !! If `filename` is not given, the file is located from `path`/`basename`/
955 !! `format`, defaulting to `this%checkpoint_path`/`checkpoint_base`/
956 !! `checkpoint_format` exactly as `optimizer_save_checkpoint` does when
957 !! saving in overwrite mode.
958 !! @param this The optimizer object.
959 !! @param filename The name of the file to load the checkpoint from.
960 !! @param design The design object.
961 !! @param path The path where the checkpoint file is located.
962 !! @param basename The base name of the file to load the checkpoint from.
963 !! @param format The file format to use for the checkpoint file.
964 !! @param iteration The iteration to load.
965 subroutine optimizer_load_checkpoint(this, design, filename, &
966 path, basename, format, iteration)
967 class(optimizer_t), intent(inout) :: this
968 class(design_t), intent(inout) :: design
969 character(len=*), intent(in), optional :: filename
970 character(len=*), intent(in), optional :: path
971 character(len=*), intent(in), optional :: basename
972 character(len=*), intent(in), optional :: format
973 integer, intent(in), optional :: iteration
974 character(len=256) :: file_full
975 character(len=12) :: suffix
976
977 call neko_log%section('Optimizer loading checkpoint')
978 call profiler_start_region('Optimizer load checkpoint')
979
980 ! Check that the provided inputs are a valid combination.
981 if (present(filename) .and. (present(path) .or. present(basename) .or. &
982 present(format) .or. present(iteration))) then
983 call neko_error('optimizer: Cannot specify both filename and ' // &
984 'path/basename/format for checkpoint loading.')
985 else if (present(filename)) then
986 file_full = trim(filename)
987 else
988 file_full = checkpoint_filename(this, path, basename, format, iteration)
989 end if
990
991 if (pe_rank .eq. 0) then
992 block
993 logical :: exists
994 character(len=256) :: msg
995 inquire(file = file_full, exist = exists)
996 if (.not. exists) then
997 write(msg, '(3A)') 'optimizer: Checkpoint file "', trim(file_full), &
998 '" does not exist.'
999 call neko_error(msg)
1000 end if
1001 end block
1002 end if
1003 call mpi_barrier(neko_comm)
1004
1005 ! Get the file extension
1006 call filename_suffix(trim(file_full), suffix)
1007 select case (trim(suffix))
1008 case ('h5', 'hdf5', 'hf5')
1009 call optimizer_load_checkpoint_hdf5(this, trim(file_full))
1010 case default
1011 call neko_error('optimizer: Unsupported checkpoint format: "' // &
1012 trim(suffix) // '"')
1013 end select
1014
1015 call this%load_checkpoint_components(trim(file_full))
1016 call design%load_checkpoint(trim(file_full))
1017
1018 ! Set the current iteration to the loaded iteration
1019 if (pe_rank .eq. 0) then
1020 write(*,*) 'Restarted simulation from checkpoint.'
1021 write(*,*) ' Checkpoint file: "', trim(file_full), '"'
1022 write(*,*) ' Iteration : ', this%current_iteration
1023 end if
1024
1025 call profiler_end_region('Optimizer load checkpoint')
1026 call neko_log%end_section()
1027
1028 end subroutine optimizer_load_checkpoint
