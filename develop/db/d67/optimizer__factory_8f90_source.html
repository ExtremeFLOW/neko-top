54 !> @brief Factory function for the optimizer.
55 !! @details This function creates an optimizer object based on the type
56 !! specified in the JSON file under "optimizer.type".
57 !!
58 !! @param object The optimizer object to be created.
59 !! @param parameters The JSON file containing the optimizer parameters.
60 !! @param problem The problem object.
61 !! @param design The design object.
62 !! @param simulation The simulation object.
63 module subroutine optimizer_factory(object, parameters, problem, design, &
64 simulation)
65 class(optimizer_t), allocatable, intent(inout) :: object
66 type(json_file), intent(inout) :: parameters
67 class(problem_t), intent(inout) :: problem
68 class(design_t), intent(in) :: design
69 type(simulation_t), optional, intent(in) :: simulation
70
71 character(len=:), allocatable :: type
72
73 if (allocated(object)) then
74 call object%free()
75 deallocate(object)
76 end if
77
78 ! Get the type of the optimizer
79 call json_get(parameters, "optimization.solver.type", type)
80
81 ! Select the optimizer type
82 select case (trim(type))
83 case ("mma")
84 allocate(mma_optimizer_t::object)
85 case ("steepest_descent")
86 allocate(steepest_descent_optimizer_t::object)
87 case default
88 call neko_type_error("Optimizer", type, KNOWN_TYPES)
89 end select
90
91 call object%init_from_json(parameters, problem, design, simulation)
92
93 end subroutine optimizer_factory
