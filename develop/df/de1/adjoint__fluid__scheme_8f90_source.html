307 !> Build the Gauss over-integration stack on first request.
308 !! @details Constructs `Xh_GL`, `dm_Xh_GL`, `gs_Xh_GL`, `c_Xh_GL`,
309 !! `GLL_to_GL` and `scratch_GL`. The stack is only needed by consumers that
310 !! over-integrate, so it is built on demand rather than in `init_base`.
311 !! The routine is idempotent: a repeated call is a no-op, leaving any
312 !! pointers already captured into the stack valid.
313 !! @note `dofmap_t%init`, `gs_t%init` and `coef_t%init` are collective, so
314 !! every call site must be reached by all ranks.
315 !! @note Must be called after `init_base`, which associates `msh` and
316 !! initialises `Xh`; both are read here.
317 !! @param this The adjoint fluid scheme owning the stack.
318 subroutine adjoint_fluid_scheme_require_gauss_stack(this)
319 class(adjoint_fluid_scheme_t), target, intent(inout) :: this
320 integer :: lxd
321
322 if (this%gauss_initialised) return
323
324 if (.not. associated(this%msh)) then
325 call neko_error('require_gauss_stack called before init_base')
326 end if
327
328 ! over intergration order (hard coded now, should be optional)
329 lxd = (3 * (this%Xh%lx + 1)) / 2
330
331 call this%Xh_GL%init(gl, lxd, lxd, lxd)
332
333 call this%dm_Xh_GL%init(this%msh, this%Xh_GL)
334
335 call this%gs_Xh_GL%init(this%dm_Xh_GL)
336
337 call this%c_Xh_GL%init(this%gs_Xh_GL)
338
339 call this%GLL_to_GL%init(this%Xh_GL, this%Xh)
340
341 ! Overintegration scratch registry (5 should be sufficient)
342 call this%scratch_GL%init(5, 2, this%dm_Xh_GL)
343
344 this%gauss_initialised = .true.
345
346 end subroutine adjoint_fluid_scheme_require_gauss_stack
