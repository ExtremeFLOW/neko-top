124 subroutine augmented_lagrangian_init_json_sim(this, json, design, simulation)
125 class(augmented_lagrangian_objective_t), intent(inout) :: this
126 type(json_file), intent(inout) :: json
127 class(design_t), intent(in) :: design
128 type(simulation_t), target, intent(inout) :: simulation
129
130 character(len=:), allocatable :: name
131 character(len=:), allocatable :: mask_name
132 real(kind=rp) :: weight
133 logical :: dealias, dealias_fallback
134
135 call json_get_or_default(json, "weight", weight, 1.0_rp)
136 call json_get_or_default(json, "mask_name", mask_name, "")
137 call json_get_or_default(json, "name", name, "Augmented Lagrangian")
138 call json_get(simulation%neko_case%params, 'case.numerics.dealias', &
139 dealias_fallback)
140 call json_get_or_default(json, "dealias_sensitivity", dealias, &
141 dealias_fallback)

143 call this%init_from_attributes(design, simulation, weight, name, &
144 mask_name, dealias)

147 !> The actual constructor.
148 !! @param this the objective.
149 !! @param design the design.
150 !! @param simulation the simulation.
151 !! @param weight the weight of the objective function.
152 !! @param name the name of the objective.
153 !! @param mask_name the name of the mask.
154 !! @param dealias should dealiasing be applied.
155 subroutine augmented_lagrangian_init_attributes(this, design, simulation, &
156 weight, name, mask_name, dealias)
157 class(augmented_lagrangian_objective_t), intent(inout) :: this
158 class(design_t), intent(in) :: design
159 type(simulation_t), target, intent(inout) :: simulation
160 real(kind=rp), intent(in) :: weight
161 character(len=*), intent(in) :: name
162 character(len=*), intent(in) :: mask_name
163 logical, intent(in) :: dealias
164
165 call this%free()
166
167 call this%init_base(name, design%size(), weight, mask_name)
168
169 ! Save the simulation and design
170 this%u => simulation%neko_case%fluid%u
171 this%v => simulation%neko_case%fluid%v
172 this%w => simulation%neko_case%fluid%w
173 this%adjoint_u => simulation%adjoint_case%fluid_adj%u_adj
174 this%adjoint_v => simulation%adjoint_case%fluid_adj%v_adj
175 this%adjoint_w => simulation%adjoint_case%fluid_adj%w_adj
176
177 this%dealias = dealias
178 ! GLL
179 this%c_Xh_GLL => simulation%neko_case%fluid%c_Xh
180 this%Xh_GLL => this%c_Xh_GLL%Xh
181
182 ! The over-integration stack is only needed if this objective dealiases
183 if (dealias) then
184 call simulation%adjoint_case%fluid_adj%require_gauss_stack()
185 end if
186
187 ! GL
188 this%c_Xh_GL => simulation%adjoint_case%fluid_adj%c_Xh_GL
189 this%Xh_GL => this%c_Xh_GL%Xh
190
191 ! GLL to GL
192 this%GLL_to_GL => simulation%adjoint_case%fluid_adj%GLL_to_GL
193
194 this%scratch_GL => simulation%adjoint_case%fluid_adj%scratch_GL
195
196 end subroutine augmented_lagrangian_init_attributes

230 !> update_value the sensitivity of the objective function with respect to
231 !! \f\f$\chi\f\f$
232 !! @param this the objective.
233 !! @param design the design.
234 subroutine augmented_lagrangian_update_sensitivity(this, design)
235 class(augmented_lagrangian_objective_t), intent(inout) :: this
236 class(design_t), intent(in) :: design
237 type(field_t), pointer :: work
238 integer :: temp_indices(1)
239 integer :: n_GL, nel
240 type(field_t), pointer :: accumulate, fld_GL, adjoint_fld_GL
241 integer :: temp_indices_GL(3)
242
243 call neko_scratch_registry%request_field(work, temp_indices(1), .false.)
244
245 if (this%dealias) then
246
247 nel = this%c_Xh_GLL%msh%nelv
248 n_gl = nel * this%Xh_GL%lxyz
249 call this%scratch_GL%request_field(accumulate, temp_indices_gl(1), &
250 .false.)
251 call this%scratch_GL%request_field(fld_gl, temp_indices_gl(2), &
252 .false.)
253 call this%scratch_GL%request_field(adjoint_fld_gl, temp_indices_gl(3), &
254 .false.)
255
256 call this%GLL_to_GL%map(fld_gl%x, this%u%x, nel, this%Xh_GL)
257 call this%GLL_to_GL%map(adjoint_fld_gl%x, this%adjoint_u%x, nel, &
258 this%Xh_GL)
259 call field_col3(accumulate, fld_gl, adjoint_fld_gl)
260
261 call this%GLL_to_GL%map(fld_gl%x, this%v%x, nel, this%Xh_GL)
262 call this%GLL_to_GL%map(adjoint_fld_gl%x, this%adjoint_v%x, nel, &
263 this%Xh_GL)
264 call field_addcol3(accumulate, fld_gl, adjoint_fld_gl)
265
266 call this%GLL_to_GL%map(fld_gl%x, this%w%x, nel, this%Xh_GL)
267 call this%GLL_to_GL%map(adjoint_fld_gl%x, this%adjoint_w%x, nel, &
268 this%Xh_GL)
269 call field_addcol3(accumulate, fld_gl, adjoint_fld_gl)
270
271 ! Evaluate term on GL and preempt the GLL premultiplication
272 if (neko_bcknd_device .eq. 1) then
273 call device_col2(accumulate%x_d, this%c_Xh_GL%B_d, n_gl)
274 call this%GLL_to_GL%map(work%x, accumulate%x, nel, this%Xh_GLL)
275 call device_invcol2(work%x_d, this%c_Xh_GLL%B_d, work%size())
276 else
277 call col2(accumulate%x, this%c_Xh_GL%B, n_gl)
278 call this%GLL_to_GL%map(work%x, accumulate%x, nel, this%Xh_GLL)
279 call invcol2(work%x, this%c_Xh_GLL%B, work%size())
280 end if
281
282 call this%scratch_GL%relinquish_field(temp_indices_gl)
283 else
284 call field_col3(work, this%u, this%adjoint_u)
285 call field_addcol3(work, this%v, this%adjoint_v)
286 call field_addcol3(work, this%w, this%adjoint_w)
287 end if
288 ! but negative
289 call field_cmult(work, -1.0_rp)
290
291 if (neko_bcknd_device .eq. 1) then
292 call device_copy(this%sensitivity%x_d, work%x_d, this%sensitivity%size())
293 else
294 call copy(this%sensitivity%x, work%x, this%sensitivity%size())
295 end if
296
297 call neko_scratch_registry%relinquish_field(temp_indices)
298
299 end subroutine augmented_lagrangian_update_sensitivity
