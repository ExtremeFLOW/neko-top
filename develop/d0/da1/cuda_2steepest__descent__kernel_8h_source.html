54template <typename T>
55__global__ void sd_direction_kernel(
56 T* __restrict__ s, const T* __restrict__ x, const T xmin, const T xmax,
57 const int n) {
58
59 const int idx = blockIdx.x * blockDim.x + threadIdx.x;
60 const int str = blockDim.x * gridDim.x;
61
62 for (int i = idx; i < n; i += str) {
63 const T xi = x[i];
64 T si = -s[i];
65 if ((xi >= xmax && si > T(0.0)) || (xi <= xmin && si < T(0.0))) {
66 si = T(0.0);
67 }
68 s[i] = si;
69 }
70}

87__global__ void sd_update_move_limits_kernel(
88 T* __restrict__ delta, const T* __restrict__ x, const T* __restrict__ x1,
89 const T* __restrict__ x2, const T dec, const T inc, const T dmin,
90 const T dmax, const int n) {
91
92 const int idx = blockIdx.x * blockDim.x + threadIdx.x;
93 const int str = blockDim.x * gridDim.x;
94
95 for (int i = idx; i < n; i += str) {
96 const T d = (x[i] - x1[i]) * (x1[i] - x2[i]);
97 T di = delta[i];
98 if (d < T(0.0)) {
99 di = dec * di;
100 } else if (d > T(0.0)) {
101 di = inc * di;
102 }
103 delta[i] = min(dmax, max(dmin, di));
104 }
105}
