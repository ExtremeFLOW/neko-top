35!> Adjoint of the curl-curl term in the Pn/Pn pressure residual.
36!! @details The Karniadakis-Israeli-Orszag splitting of Neko's
37!! \f$ P_N-P_N \f$ scheme puts the rotational form of the viscous term,
38!! evaluated on the extrapolated velocity \f$ \mathbf{u}_e \f$, into the
39!! pressure residual (`pnpn_prs_res`). Before gather-scatter, that
40!! contribution is
41!! \f[
42!! r_{cc}(\mathbf{u}_e) = -\frac{\mu}{\rho} G^T M \mathbf{W}, \qquad
43!! \mathbf{W} = M D_c M D_c \mathbf{u}_e,
44!! \f]
45!! where \f$ D_c \f$ is the element-local, pointwise curl,
46!! \f$ M \mathbf{y} = \bar{B}^{-1} \mathrm{gs}(B \mathbf{y}) \f$ is the
47!! mass-weighted average that Neko's `curl` applies to it (so that
48!! `curl` is \f$ M D_c \f$), with \f$ B \f$ the element-local diagonal mass
49!! matrix and \f$ \bar{B} = \mathrm{gs}(B) \f$ the assembled one (`coef%Binv`
50!! holds \f$ \bar{B}^{-1} \f$), and \f$ G \f$ is the weak gradient `opgrad`,
51!! \f$ G_i p = B D_i p \f$, whose transpose is `cdtp`. Since
52!! \f$ \bar{B} = \mathrm{gs}(B) \f$, \f$ M \f$ maps every field to a
53!! continuous one and leaves continuous fields unchanged, so it is a
54!! projection, \f$ M^2 = M \f$, and
55!! \f[
56!! r_{cc}(\mathbf{u}_e) = -\frac{\mu}{\rho} G^T M D_c M D_c \mathbf{u}_e .
57!! \f]
58!! The adjoint of this term with respect to \f$ \mathbf{u}_e \f$, acting on
59!! the adjoint pressure \f$ p^\dagger \f$, therefore carries two factors
60!! \f$ M^T \f$ and is the volume load
61!! \f[
62!! \mathbf{L} = \frac{\mu}{\rho} D_c^T M^T D_c^T M^T G p^\dagger, \qquad
63!! M^T \mathbf{y} = B \, \mathrm{gs}(\bar{B}^{-1} \mathbf{y}),
64!! \f]
65!! i.e. for every continuous \f$ \mathbf{u}_e \f$ and \f$ p^\dagger \f$,
66!! \f$ \mathbf{u}_e \cdot \mathbf{L} = -p^\dagger \cdot
67!! r_{cc}(\mathbf{u}_e) \f$, with both dot products summed over all local
68!! (non-unique) nodes and neither \f$ \mathbf{L} \f$ nor \f$ r_{cc} \f$
69!! gather-scattered.
70!!
71!! The transpose is exact in three dimensions only. For a two-dimensional
72!! mesh Neko's `curl` drops the \f$ z \f$ derivatives and does not
73!! mass-average the \f$ z \f$ component, whereas \f$ \mathbf{L} \f$ applies
74!! the full three-dimensional \f$ D_c^T \f$ and \f$ M^T \f$ to all three
75!! components. The adjoint case therefore rejects two-dimensional meshes,
76!! and `adjoint_curl_curl_check` rejects the boundary conditions for which
77!! the load is not the transpose of the primal term.
78!!
79!! In the continuous setting \f$ \nabla \times \nabla p^\dagger = 0 \f$ and
80!! the term reduces to a boundary integral. The discrete operators do not
81!! satisfy that identity, so a boundary-only form is not the transpose of
82!! what the primal computes; the load acts on the whole volume.

105 !> Stop with an error if the boundary conditions make the adjoint
106 !! curl-curl load differ from the transpose of the primal's curl-curl
107 !! pressure term.
108 !! @details The load does not transpose the symmetry-surface part of the
109 !! primal term (`bc_sym_surface`), nor the cyclic rotations applied inside
110 !! the primal's `curl`. The symmetry test is reduced over all ranks, so
111 !! that every rank stops, also those that hold no symmetry facet. The
112 !! restriction to three-dimensional meshes (see the module description) is
113 !! not tested here: it has to be enforced before the adjoint fluid is
114 !! initialised, which assumes a three-dimensional mesh.
115 !! @param prim_sym The symmetry surface of the primal pressure residual.
116 !! @param adj_sym The symmetry surface of the adjoint pressure residual.
117 !! @param cyclic Whether the primal has cyclic periodic boundaries.
118 subroutine adjoint_curl_curl_check(prim_sym, adj_sym, cyclic)
119 type(facet_normal_t), intent(in) :: prim_sym, adj_sym
120 logical, intent(in) :: cyclic
121 logical :: has_sym
122
123 has_sym = prim_sym%marked_facet%size() .gt. 0 .or. &
124 adj_sym%marked_facet%size() .gt. 0
125 call mpi_allreduce(mpi_in_place, has_sym, 1, mpi_logical, mpi_lor, &
126 neko_comm)
127
128 if (has_sym) then
129 call neko_error("Symmetry boundaries " // &
130 "(case.fluid.boundary_conditions or " // &
131 "case.adjoint_fluid.boundary_conditions) are not supported " // &
132 "by the adjoint: the transpose of the curl-curl pressure " // &
133 "term on symmetry surfaces is not implemented.")
134 end if
135
136 if (cyclic) then
137 call neko_error("Cyclic boundaries (case.fluid.cyclic) are not " // &
138 "supported by the adjoint: the transpose of the cyclic " // &
139 "rotations in the curl-curl pressure term is not implemented.")
140 end if
141
142 end subroutine adjoint_curl_curl_check

144 !> Compute the adjoint curl-curl load
145 !! \f$ \mathbf{L} = \frac{\mu}{\rho} D_c^T M^T D_c^T M^T G p^\dagger \f$.
146 !! @details See the module description for the notation. The load is
147 !! element-local (not gather-scattered), since it is added to the adjoint
148 !! forcing, which the velocity residual gather-scatters. It is added to the
149 !! forcing before `makeabf`, which scales the forcing by \f$ \rho \f$ and
150 !! extrapolates it in time, so the velocity right-hand side receives
151 !! \f$ \mu D_c^T M^T D_c^T M^T G p^\dagger \f$, weighted over the lagged
152 !! adjoint pressures as the primal's extrapolation weights
153 !! \f$ \mathbf{u}_e \f$. The material properties are taken as constant, as
154 !! in the primal residual. The cyclic rotations and the symmetry-surface
155 !! part of the primal term are not transposed, so the load is not valid for
156 !! cyclic or symmetry boundaries (see `adjoint_curl_curl_check`), nor for a
157 !! two-dimensional mesh. All field arguments must be distinct.
158 !! @param load_x x component of the load.
159 !! @param load_y y component of the load.
160 !! @param load_z z component of the load.
161 !! @param p_adj The adjoint pressure (continuous).
162 !! @param coef The SEM coefficients.
163 !! @param gs The gather-scatter operator of the velocity space.
164 !! @param mu The constant dynamic viscosity.
165 !! @param rho The constant density.
166 !! @param work1 Work field.
167 !! @param work2 Work field.
168 !! @param work3 Work field.
169 !! @param work4 Work field.
170 subroutine adjoint_curl_curl_load(load_x, load_y, load_z, p_adj, coef, gs, &
171 mu, rho, work1, work2, work3, work4)
172 type(field_t), intent(inout) :: load_x, load_y, load_z
173 type(field_t), intent(in) :: p_adj
174 type(coef_t), intent(in) :: coef
175 type(gs_t), intent(inout) :: gs
176 real(kind=rp), intent(in) :: mu, rho
177 type(field_t), intent(inout) :: work1, work2, work3, work4
178
179 ! g = Binv gs(G p), so that B g = M^T G p; held in load
180 call opgrad(load_x%x, load_y%x, load_z%x, p_adj%x, coef)
181 call adjoint_curl_curl_gs_binv(load_x, load_y, load_z, coef, gs)
182
183 ! c = Binv gs(D_c^T B g), so that B c = M^T D_c^T M^T G p; held in
184 ! work1..3
185 call adjoint_curl_transpose(work1, work2, work3, load_x, load_y, load_z, &
186 work4, coef)
187 call adjoint_curl_curl_gs_binv(work1, work2, work3, coef, gs)
188
189 ! L = (mu / rho) D_c^T B c
190 call adjoint_curl_transpose(load_x, load_y, load_z, work1, work2, work3, &
191 work4, coef)
192 call field_cmult(load_x, mu / rho)
193 call field_cmult(load_y, mu / rho)
194 call field_cmult(load_z, mu / rho)
195
196 end subroutine adjoint_curl_curl_load

198 !> Apply the transpose of the pointwise curl to mass-weighted data,
199 !! \f$ \mathbf{w} = D_c^T B \mathbf{y} \f$, element by element (no
200 !! gather-scatter): \f$ w_x = (B D_z)^T y_y - (B D_y)^T y_z \f$ and
201 !! cyclically.
202 !! @param w1 x component of the result.
203 !! @param w2 y component of the result.
204 !! @param w3 z component of the result.
205 !! @param y1 x component of the input.
206 !! @param y2 y component of the input.
207 !! @param y3 z component of the input.
208 !! @param work Work field.
209 !! @param coef The SEM coefficients.
210 subroutine adjoint_curl_transpose(w1, w2, w3, y1, y2, y3, work, coef)
211 type(field_t), intent(inout) :: w1, w2, w3
212 type(field_t), intent(inout) :: y1, y2, y3
213 type(field_t), intent(inout) :: work
214 type(coef_t), intent(in) :: coef
215
216 call cdtp(w1%x, y2%x, coef%drdz, coef%dsdz, coef%dtdz, coef)
217 call cdtp(work%x, y3%x, coef%drdy, coef%dsdy, coef%dtdy, coef)
218 call field_sub2(w1, work)
219
220 call cdtp(w2%x, y3%x, coef%drdx, coef%dsdx, coef%dtdx, coef)
221 call cdtp(work%x, y1%x, coef%drdz, coef%dsdz, coef%dtdz, coef)
222 call field_sub2(w2, work)
223
224 call cdtp(w3%x, y1%x, coef%drdy, coef%dsdy, coef%dtdy, coef)
225 call cdtp(work%x, y2%x, coef%drdx, coef%dsdx, coef%dtdx, coef)
226 call field_sub2(w3, work)
227
228 end subroutine adjoint_curl_transpose

230 !> Apply \f$ \mathbf{y} \leftarrow \bar{B}^{-1} \mathrm{gs}(\mathbf{y}) \f$
231 !! (`coef%Binv` times the gather-scatter sum) to the three components of a
232 !! vector field.
233 !! @param y1 x component.
234 !! @param y2 y component.
235 !! @param y3 z component.
236 !! @param coef The SEM coefficients.
237 !! @param gs The gather-scatter operator.
238 subroutine adjoint_curl_curl_gs_binv(y1, y2, y3, coef, gs)
239 type(field_t), intent(inout) :: y1, y2, y3
240 type(coef_t), intent(in) :: coef
241 type(gs_t), intent(inout) :: gs
242 integer :: n
243
244 n = y1%size()
245
246 call gs%op(y1, gs_op_add)
247 call gs%op(y2, gs_op_add)
248 call gs%op(y3, gs_op_add)
249
250 if (neko_bcknd_device .eq. 1) then
251 call device_col2(y1%x_d, coef%Binv_d, n)
252 call device_col2(y2%x_d, coef%Binv_d, n)
253 call device_col2(y3%x_d, coef%Binv_d, n)
254 else
255 call col2(y1%x, coef%Binv, n)
256 call col2(y2%x, coef%Binv, n)
257 call col2(y3%x, coef%Binv, n)
258 end if
259
260 end subroutine adjoint_curl_curl_gs_binv
