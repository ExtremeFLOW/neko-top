35!> Implements the `viscous_dissipation_objective_t` type.
36!
37! I promise I'll write this document properly in the future...
38!
39! But the Borrvall & Petersson (I think) paper had an objective function
40! that had 2 terms, dissipation and this term they claimed represented
41! out of plane stresses.
42! I never really understood that extra term, I also don't think it
43! applies to 3D cases, but everyone includes it anyway.
44!
45! It appears to me to be basically a heuristic penality that targets
46! non-binary designs
47!
48! so let's call
49!
50! F = \int |\nabla u|^2 + K \int \chi \u^2
51!
52! | dissipation | |"Brinkman dissipation"|
53!
54! I say "Brinkman dissipation" because they said it came from lubrication
55! theory...
56! Anyway, we can change all this later (especially the names!)
57
58! If the objective function \frac{\mu}{2} \int |\nabla u|^2,
59! the corresponding adjoint forcing is \mu \int \nabla v \cdot \nabla u
60!
61! for the Brinkman dissipation, the adjoint forcing is \chi u
62!
63! This has always annoyed me...
64! because now I see one objective and one constraint
65!

249 !> Compute the objective function.
250 !! @param this the objective.
251 !! @param design the design.
252 subroutine viscous_dissipation_update_value(this, design)
253 class(viscous_dissipation_objective_t), intent(inout) :: this
254 class(design_t), intent(in) :: design
255 type(field_t), pointer :: wo1, wo2, wo3, work
256 type(field_t), pointer :: objective_field
257 integer :: temp_indices(5)
258 integer n
259
260 call neko_scratch_registry%request_field(wo1, temp_indices(1), .false.)
261 call neko_scratch_registry%request_field(wo2, temp_indices(2), .false.)
262 call neko_scratch_registry%request_field(wo3, temp_indices(3), .false.)
263 call neko_scratch_registry%request_field(objective_field, temp_indices(4), &
264 .false.)
265 call neko_scratch_registry%request_field(work, temp_indices(5), .false.)
266
267 ! update_value the objective function.
268 call grad(wo1%x, wo2%x, wo3%x, this%u%x, this%c_Xh)
269 call field_col3(objective_field, wo1, wo1)
270 call field_addcol3(objective_field, wo2, wo2)
271 call field_addcol3(objective_field, wo3, wo3)
272
273 call grad(wo1%x, wo2%x, wo3%x, this%v%x, this%c_Xh)
274 call field_addcol3(objective_field, wo1, wo1)
275 call field_addcol3(objective_field, wo2, wo2)
276 call field_addcol3(objective_field, wo3, wo3)
277
278 call grad(wo1%x, wo2%x, wo3%x, this%w%x, this%c_Xh)
279 call field_addcol3(objective_field, wo1, wo1)
280 call field_addcol3(objective_field, wo2, wo2)
281 call field_addcol3(objective_field, wo3, wo3)
282
283 ! integrate the field
284 n = wo1%size()
285 if (this%has_mask) then
286 if (neko_bcknd_device .eq. 1) then
287 ! note, this could be done more elagantly by writing
288 ! device_glsc2_mask
289 call field_copy(work, objective_field)
290 call mask_exterior_const(work, this%mask, 0.0_rp)
291 this%value = device_glsc2(work%x_d, this%c_xh%B_d, n)
292 else
293 this%value = glsc2_mask(objective_field%x, this%c_Xh%b, &
294 n, this%mask%mask%get(), this%mask%size)
295 end if
296 else
297 if (neko_bcknd_device .eq. 1) then
298 this%value = device_glsc2(objective_field%x_d, &
299 this%c_Xh%b_d, n)
300 else
301 this%value = glsc2(objective_field%x, this%c_Xh%b, n)
302 end if
303 end if
304
305 this%value = this%value * 0.5_rp * this%viscosity / this%volume
306
307 call neko_scratch_registry%relinquish_field(temp_indices)
308
309 end subroutine viscous_dissipation_update_value
