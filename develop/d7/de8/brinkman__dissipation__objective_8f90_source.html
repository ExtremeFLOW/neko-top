129 !> The common constructor using a JSON object.
130 !! @param this The objective.
131 !! @param json the JSON object.
132 !! @param design the design.
133 !! @param simulation the simulation.
134 subroutine brinkman_dissipation_init_json_sim(this, json, design, simulation)
135 class(brinkman_dissipation_objective_t), intent(inout) :: this
136 type(json_file), intent(inout) :: json
137 class(design_t), intent(in) :: design
138 type(simulation_t), target, intent(inout) :: simulation
139
140 character(len=:), allocatable :: mask_name
141 character(len=:), allocatable :: name
142 real(kind=rp) :: weight
143 logical :: dealias_sensitivity, dealias_forcing, dealias_fallback
144 real(kind=rp) :: start_time, end_time
145
146 call nekotop_continuation%json_get_or_register(json, 'weight', &
147 this%weight, weight, 1.0_rp)
148 call json_get_or_default(json, "mask_name", mask_name, "")
149 call json_get_or_default(json, "name", name, "Brinkman dissipation")
150 call json_get(simulation%neko_case%params, 'case.numerics.dealias', &
151 dealias_fallback)
152 call json_get_or_default(json, "dealias_sensitivity", &
153 dealias_sensitivity, dealias_fallback)
154 call json_get_or_default(json, "dealias_forcing", &
155 dealias_forcing, dealias_fallback)
156 call json_get_or_default(json, "start_time", start_time, 0.0_rp)
157 call json_get_or_default(json, "end_time", end_time, huge(0.0_rp))
158
159 ! The two keys select one quadrature. Set differently, the sensitivity
160 ! is not the gradient of any value this objective could report.
161 if (dealias_sensitivity .neqv. dealias_forcing) then
162 call neko_error("brinkman_dissipation: dealias_sensitivity and " // &
163 "dealias_forcing must be equal")
164 end if
165
166 call this%init_from_attributes(design, simulation, weight, name, &
167 mask_name, dealias_forcing, start_time, end_time)
168 end subroutine brinkman_dissipation_init_json_sim

170 !> The actual constructor.
171 !! @param this The objective.
172 !! @param design the design.
173 !! @param simulation the simulation.
174 !! @param weight the weight of the objective function.
175 !! @param name the name of the objective.
176 !! @param mask_name the name of the mask.
177 !! @param dealias integrate the value, the adjoint forcing and the
178 !! sensitivity on the GL points.
179 !! @param start_time start of the integration window.
180 !! @param end_time end of the integration window.
181 subroutine brinkman_dissipation_init_attributes(this, design, simulation, &
182 weight, name, mask_name, dealias, start_time, end_time)
183 class(brinkman_dissipation_objective_t), intent(inout) :: this
184 class(design_t), intent(in) :: design
185 type(simulation_t), target, intent(inout) :: simulation
186 real(kind=rp), intent(in) :: weight
187 character(len=*), intent(in) :: name
188 character(len=*), intent(in) :: mask_name
189 logical, intent(in) :: dealias
190 real(kind=rp), intent(in) :: start_time
191 real(kind=rp), intent(in) :: end_time
192
193 type(adjoint_brinkman_dissipation_source_term_t) :: brinkman_dissipation
194 real(kind=rp) :: forcing_start, forcing_end
195
196 call this%free()
197
198 ! Call the base initializer
199 call this%init_base(name, design%size(), weight, mask_name, &
200 start_time, end_time)
201
202 this%dealias = dealias
203
204 ! Grab the Brinkman amplitude field.
205 select type (design)
206 type is (brinkman_design_t)
207 this%brinkman_amplitude => &
208 neko_registry%get_field("brinkman_amplitude")
209
210
211 class default
212 call neko_error('Brinkman dissipation only works with '// &
213 'brinkman_design')
214 end select
215
216 this%u => neko_registry%get_field('u')
217 this%v => neko_registry%get_field('v')
218 this%w => neko_registry%get_field('w')
219
220 ! GLL
221 this%c_Xh_GLL => simulation%neko_case%fluid%c_Xh
222 this%Xh_GLL => simulation%neko_case%fluid%c_Xh%Xh
223 this%gdim = this%c_Xh_GLL%msh%gdim
224
225 ! The over-integration stack is only needed if this objective dealiases
226 if (dealias) then
227 call simulation%adjoint_case%fluid_adj%require_gauss_stack()
228 end if
229
230 ! GL
231 this%c_Xh_GL => simulation%adjoint_case%fluid_adj%c_Xh_GL
232 this%Xh_GL => this%c_Xh_GL%Xh
233
234 ! GLL to GL
235 this%GLL_to_GL => simulation%adjoint_case%fluid_adj%GLL_to_GL
236
237 this%scratch_GL => simulation%adjoint_case%fluid_adj%scratch_GL
238
239 ! compute the volume of the objective domain
240 if (this%has_mask) then
241 this%volume = compute_masked_volume(this%mask, this%c_Xh_GLL)
242 else
243 this%volume = this%c_Xh_GLL%volume
244 end if
245
246 ! if we have the Brinkman dissipation we initialize and append it too
247 call this%get_forcing_window(simulation, forcing_start, forcing_end)
248
249 associate(f_adj_x => simulation%adjoint_fluid%f_adj_x, &
250 f_adj_y => simulation%adjoint_fluid%f_adj_y, &
251 f_adj_z => simulation%adjoint_fluid%f_adj_z)
252
253 call brinkman_dissipation%init_from_components(f_adj_x, f_adj_y, &
254 f_adj_z, design, this%weight, this%u, this%v, this%w, this%mask, &
255 this%has_mask, this%c_Xh_GLL, this%c_Xh_GL, this%GLL_to_GL, &
256 this%dealias, this%volume, this%scratch_GL, forcing_start, &
257 forcing_end, this%forcing_scale)
258
259 end associate
260
261 ! append adjoint forcing term based on objective function
262 select type (f => simulation%adjoint_fluid)
263 type is (adjoint_fluid_pnpn_t)
264 call f%source_term%add_source_term(brinkman_dissipation)
265 end select
266
267 end subroutine brinkman_dissipation_init_attributes

287 !> Compute the objective function.
288 !! With dealiasing the integrand is built on the GL points exactly as the
289 !! adjoint forcing and the sensitivity differentiate it, from the masked
290 !! \f$\chi\f$ interpolated to GL.
291 !! @param this The objective.
292 !! @param design the design.
293 subroutine brinkman_dissipation_update_value(this, design)
294 class(brinkman_dissipation_objective_t), intent(inout) :: this
295 class(design_t), intent(in) :: design
296 type(field_t), pointer :: work
297 integer :: temp_indices(1)
298 integer :: n_GL, nel
299 type(field_t), pointer :: accumulate, fld_GL, chi_GL
300 integer :: temp_indices_GL(3)
301
302 call neko_scratch_registry%request_field(work, temp_indices(1), .false.)
303
304 if (this%dealias) then
305 nel = this%c_Xh_GLL%msh%nelv
306 n_gl = nel * this%Xh_GL%lxyz
307 call this%scratch_GL%request_field(accumulate, temp_indices_gl(1), &
308 .false.)
309 call this%scratch_GL%request_field(fld_gl, temp_indices_gl(2), .false.)
310 call this%scratch_GL%request_field(chi_gl, temp_indices_gl(3), .false.)
311
312 call field_copy(work, this%brinkman_amplitude)
313 if (this%has_mask) then
314 call mask_exterior_const(work, this%mask, 0.0_rp)
315 end if
316 call this%GLL_to_GL%map(chi_gl%x, work%x, nel, this%Xh_GL)
317
318 call this%GLL_to_GL%map(fld_gl%x, this%u%x, nel, this%Xh_GL)
319 call field_col3(accumulate, fld_gl, fld_gl)
320 call this%GLL_to_GL%map(fld_gl%x, this%v%x, nel, this%Xh_GL)
321 call field_addcol3(accumulate, fld_gl, fld_gl)
322 if (this%gdim .eq. 3) then
323 call this%GLL_to_GL%map(fld_gl%x, this%w%x, nel, this%Xh_GL)
324 call field_addcol3(accumulate, fld_gl, fld_gl)
325 end if
326 call field_col2(accumulate, chi_gl)
327
328 if (neko_bcknd_device .eq. 1) then
329 this%value = device_glsc2(accumulate%x_d, this%c_Xh_GL%B_d, n_gl)
330 else
331 this%value = glsc2(accumulate%x, this%c_Xh_GL%B, n_gl)
332 end if
333
334 call this%scratch_GL%relinquish_field(temp_indices_gl)
335 else
336 call field_col3(work, this%u, this%u)
337 call field_addcol3(work, this%v, this%v)
338 if (this%gdim .eq. 3) then
339 call field_addcol3(work, this%w, this%w)
340 end if
341 call field_col2(work, this%brinkman_amplitude)
342
343 if (this%has_mask) then
344 if (neko_bcknd_device .eq. 1) then
345 ! note, this could be done more elagantly by writing
346 ! device_glsc2_mask
347 call mask_exterior_const(work, this%mask, 0.0_rp)
348 this%value = device_glsc2(work%x_d, this%c_Xh_GLL%B_d, &
349 design%size())
350 else
351 this%value = glsc2_mask(work%x, this%c_Xh_GLL%B, design%size(), &
352 this%mask%mask%get(), this%mask%size)
353 end if
354 else
355 if (neko_bcknd_device .eq. 1) then
356 this%value = device_glsc2(work%x_d, this%c_Xh_GLL%B_d, &
357 design%size())
358 else
359 this%value = glsc2(work%x, this%c_Xh_GLL%B, design%size())
360 end if
361 end if
362 end if
363 this%value = 0.5_rp * this%value / this%volume
364
365 call neko_scratch_registry%relinquish_field(temp_indices)
366
367 end subroutine brinkman_dissipation_update_value

369 !> update_value the sensitivity of the objective function with respect to
370 !! \f$chi\f$
371 !! @param this The objective.
372 !! @param design the design.
373 subroutine brinkman_dissipation_update_sensitivity(this, design)
374 class(brinkman_dissipation_objective_t), intent(inout) :: this
375 class(design_t), intent(in) :: design
376 type(field_t), pointer :: work
377 integer :: temp_indices(1)
378 integer :: n_GL, nel
379 type(field_t), pointer :: accumulate, fld_GL
380 integer :: temp_indices_GL(2)
381
382 ! The Brinkman dissipation adds an extra term in the sensitivity.
383
384 call neko_scratch_registry%request_field(work, temp_indices(1), .false.)
385
386 if (this%dealias) then
387 nel = this%c_Xh_GLL%msh%nelv
388 n_gl = nel * this%Xh_GL%lxyz
389 call this%scratch_GL%request_field(accumulate, temp_indices_gl(1), &
390 .false.)
391 call this%scratch_GL%request_field(fld_gl, temp_indices_gl(2), .false.)
392
393 call this%GLL_to_GL%map(fld_gl%x, this%u%x, nel, this%Xh_GL)
394 call field_col3(accumulate, fld_gl, fld_gl)
395 call this%GLL_to_GL%map(fld_gl%x, this%v%x, nel, this%Xh_GL)
396 call field_addcol3(accumulate, fld_gl, fld_gl)
397 if (this%gdim .eq. 3) then
398 call this%GLL_to_GL%map(fld_gl%x, this%w%x, nel, this%Xh_GL)
399 call field_addcol3(accumulate, fld_gl, fld_gl)
400 end if
401 ! scale
402 call field_cmult(accumulate, this%weight * 0.5_rp / this%volume)
403
404 ! Evaluate term on GL and preempt the GLL premultiplication
405 if (neko_bcknd_device .eq. 1) then
406 call device_col2(accumulate%x_d, this%c_Xh_GL%B_d, n_gl)
407 call this%GLL_to_GL%map(work%x, accumulate%x, nel, this%Xh_GLL)
408 call device_invcol2(work%x_d, this%c_Xh_GLL%B_d, work%size())
409 else
410 call col2(accumulate%x, this%c_Xh_GL%B, n_gl)
411 call this%GLL_to_GL%map(work%x, accumulate%x, nel, this%Xh_GLL)
412 call invcol2(work%x, this%c_Xh_GLL%B, work%size())
413 end if
414
415 call this%scratch_GL%relinquish_field(temp_indices_gl)
416
417 else
418 call field_col3(work, this%u, this%u)
419 call field_addcol3(work, this%v, this%v)
420 if (this%gdim .eq. 3) then
421 call field_addcol3(work, this%w, this%w)
422 end if
423 ! scale
424 call field_cmult(work, this%weight * 0.5_rp / this%volume)
425 end if
426
427 ! chi enters the value only inside the mask, so the sensitivity is zero
428 ! outside it
429 if (this%has_mask) then
430 call mask_exterior_const(work, this%mask, 0.0_rp)
431 end if
432
433 if (neko_bcknd_device .eq. 1) then
434 call device_copy(this%sensitivity%x_d, work%x_d, this%sensitivity%size())
435 else
436 call copy(this%sensitivity%x, work%x, this%sensitivity%size())
437 end if
438
439 call neko_scratch_registry%relinquish_field(temp_indices)
440
441 end subroutine brinkman_dissipation_update_sensitivity
