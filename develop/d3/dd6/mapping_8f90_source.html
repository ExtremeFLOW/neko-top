152 !> Constructor for the `mapping_t` (base) class.
153 !! @param this The mapping object.
154 !! @param json The JSON with properties.
155 !! @param coef The SEM coefficients.
156 !! @param fld_name Registry name of the field storing the unmapped input.
157 !! If it is taken, e.g. by a second mapping of the same type in one run,
158 !! the first free name of the form `fld_name_2`, `fld_name_3`, ... is used.
159 !! @note Neko's registry cannot remove a field and `free` does not release
160 !! it, so a re-initialised mapping also takes the next free suffix, while a
161 !! lookup of the base name still returns the field of the first, stale
162 !! instance. Lookups by name, e.g. in examples/wiggles/user.f90 or through
163 !! the checkpoint `extra_fields`, are reliable only for the first
164 !! initialisation.
165 subroutine mapping_init_base(this, json, coef, fld_name)
166 class(mapping_t), intent(inout) :: this
167 type(json_file), intent(inout) :: json
168 character(len=*), intent(in) :: fld_name
169 type(coef_t), intent(inout), target :: coef
170 integer :: i
171
172 this%coef => coef
173
174 ! Each instance needs its own field, since the backward mapping uses the
175 ! input stored by the forward mapping of the same instance.
176 this%fld_name = fld_name
177 i = 1
178 do while (neko_registry%field_exists(trim(this%fld_name)))
179 i = i + 1
180 write(this%fld_name, '(A,"_",I0)') trim(fld_name), i
181 end do
182
183 ! The mapping handler will take care of naming this field
184 call neko_registry%add_field(coef%dof, this%fld_name)
185 this%X_in => neko_registry%get_field(this%fld_name)
186
187 end subroutine mapping_init_base
188

191 class(mapping_t), intent(inout) :: this
192
193 nullify(this%X_in)
194 nullify(this%coef)
195
196 end subroutine mapping_free_base

208 !> Executes `apply_forward`
209 !! @param this The mapping object
210 !! @param X_out The mapped field (\f$\tilde{\rho}\f$)
211 !! @param X_in The unmapped field (\f$\rho\f$)
212 subroutine mapping_apply_forward_wrapper(this, X_out, X_in)
213 class(mapping_t), intent(inout) :: this
214 type(field_t), intent(inout) :: X_out
215 type(field_t), intent(in) :: X_in
216
217 call field_copy(this%X_in, x_in)
218 call this%forward_mapping(x_out, this%X_in)

220 end subroutine mapping_apply_forward_wrapper
