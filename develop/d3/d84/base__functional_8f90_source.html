51 !> The base functional type
52 !!
53 !! This is the base class for objectives and constraints alike.
54 !! A base functional should be able to evaluate itself and its sensitivity
55 !! with respect to the design variables.
56 !!
57 !! The base functional is also responsible for managing the adjoint forcing
58 !! terms that are required for the adjoint problem, any source terms
59 !! simulation components that are required to evaluate the base functional.
60 !! All of which should be prepared in the `init` method.
61 type, abstract, public :: base_functional_t

271 !> Accumulate the value of the function
272 !!
273 !! Updates the value in place as the running mean of the function.
274 !!
275 !! The result is the running mean of the function over the part of the
276 !! functional's own time window that has been simulated so far, normalised
277 !! by the length of time actually sampled. It is therefore invariant to the
278 !! length of the run: extending `end_time` of the simulation past the
279 !! functional's window leaves the value unchanged, and a windowed
280 !! functional reports the mean over its window rather than over the run.
281 !! @param this The functional.
282 !! @param design The design.
283 !! @param time The current time state.
284 subroutine functional_accumulate_value(this, design, time)
285 class(base_functional_t), intent(inout) :: this
286 class(design_t), intent(in) :: design
287 type(time_state_t), intent(in) :: time
288 real(kind=xp) :: value_new, value_old, weight_new, weight_old, dt
289
290 if (.not. this%in_window(time)) return
291
292 ! Store the old value and compute the new value.
293 value_old = real(this%value, kind=xp)
294 call this%update_value(design)
295 value_new = real(this%value, kind=xp)
296
297 ! Compute the weights for the old and new values.
298 dt = real(time%dt, kind=xp)
299 weight_new = dt / (this%value_weight + dt)
300 weight_old = this%value_weight / (this%value_weight + dt)
301
302 ! Rectangle rule; could potentially use higher order trapezoidal/Simpson
303 ! etc, but this should suffice.
304 this%value = real(value_old * weight_old + value_new * weight_new, kind=rp)
305 this%value_weight = this%value_weight + dt
306 end subroutine functional_accumulate_value
307
308 !> Accumulate the sensitivity of the function
309 !!
310 !! Normalised exactly as `accumulate_value`, so that the sensitivity stays
311 !! consistent with the value it differentiates.
312 !! @param this The functional.
313 !! @param design The design.
314 !! @param time The current time state.
315 subroutine functional_accumulate_sensitivity(this, design, time)
316 class(base_functional_t), intent(inout) :: this
317 class(design_t), intent(in) :: design
318 type(time_state_t), intent(in) :: time
319 real(kind=xp) :: weight_new, weight_old, dt
320 type(vector_t), pointer :: sensitivity_new, sensitivity_old
321 integer :: n, idx(2)
322
323 if (.not. this%in_window(time)) return
324
325 n = this%sensitivity%size()
326 call neko_scratch_registry%request(sensitivity_new, idx(1), n, .false.)
327 call neko_scratch_registry%request(sensitivity_old, idx(2), n, .false.)
328
329 ! Store old sensitivity and compute new sensitivity.
330 call vector_copy(sensitivity_old, this%sensitivity)
331 call this%update_sensitivity(design)
332 call vector_copy(sensitivity_new, this%sensitivity)
333
334 ! Compute the weights for the old and new sensitivities.
335 dt = real(time%dt, kind=xp)
336 weight_new = dt / (this%sensitivity_weight + dt)
337 weight_old = this%sensitivity_weight / (this%sensitivity_weight + dt)
338
339 ! Rectangle rule; could potentially use higher order trapezoidal/Simpson
340 ! etc, but this should suffice.
341 call vector_add3s2(this%sensitivity, &
342 sensitivity_new, sensitivity_old, weight_new, weight_old)
343 this%sensitivity_weight = this%sensitivity_weight + dt
344
345 nullify(sensitivity_new)
346 nullify(sensitivity_old)
347 call neko_scratch_registry%relinquish(idx)
348 end subroutine functional_accumulate_sensitivity

353 !> Whether a sample at the current time contributes to the average.
354 !!
355 !! A tolerance of a small fraction of the timestep is applied to both ends
356 !! of the window, so that a boundary placed exactly on a timestep is
357 !! included deterministically rather than at the mercy of the round-off
358 !! accumulated in `time%t`.
359 !! @param this The functional.
360 !! @param time The current time state.
361 !! @return Whether `time%t` lies inside the accumulation window.
362 pure function functional_in_window(this, time) result(inside)
363 class(base_functional_t), intent(in) :: this
364 type(time_state_t), intent(in) :: time
365 logical :: inside
366 real(kind=rp) :: tol
367
368 tol = 1.0e-6_rp * time%dt
369 inside = time%t .ge. this%start_time - tol .and. &
370 time%t .le. this%end_time + tol
371 end function functional_in_window
